    async def _validate_and_parse_public_key(self, public_key: str) -> Dict[str, Any]:
        """Validate and parse SSH public key format"""
        try:
            # Clean up the key
            public_key = public_key.strip()
            
            # Split key components
            parts = public_key.split()
            if len(parts) < 2:
                raise ValueError("Invalid SSH public key format")
            
            key_type = parts[0]
            key_data = parts[1]
            comment = ' '.join(parts[2:]) if len(parts) > 2 else ''
            
            # Validate key type
            supported_types = {
                'ssh-rsa': 'rsa',
                'ssh-ed25519': 'ed25519',
                'ecdsa-sha2-nistp256': 'ecdsa',
                'ecdsa-sha2-nistp384': 'ecdsa',
                'ecdsa-sha2-nistp521': 'ecdsa'
            }
            
            if key_type not in supported_types:
                raise ValueError(f"Unsupported key type: {key_type}")
            
            # Validate base64 encoding
            try:
                key_bytes = base64.b64decode(key_data)
            except Exception:
                raise ValueError("Invalid base64 encoding in key data")
            
            # Calculate fingerprint
            fingerprint = self._calculate_fingerprint(public_key.encode())
            
            # Extract key size for RSA keys
            key_size = None
            if key_type == 'ssh-rsa':
                key_size = self._extract_rsa_key_size(key_bytes)
                if key_size < self.min_rsa_key_size:
                    raise ValueError(f"RSA key size {key_size} is below minimum {self.min_rsa_key_size}")
            
            return {
                'type': supported_types[key_type],
                'data': key_data,
                'comment': comment,
                'fingerprint': fingerprint,
                'size': key_size
            }
            
        except Exception as e:
            logger.error("Public key validation failed", error=str(e))
            raise ValueError(f"Invalid SSH public key: {str(e)}")

    async def validate_ssh_key_usage(
        self, 
        fingerprint: str, 
        source_ip: str,
        repository: Optional[str] = None
    ) -> bool:
        """Validate SSH key usage with IP restrictions and rate limiting"""
        try:
            # Check cache first
            cache_key = f"{fingerprint}:{source_ip}"
            cached_result = self.validation_cache.get(cache_key)
            
            if cached_result and time.time() - cached_result['timestamp'] < self.cache_ttl:
                return cached_result['valid']
            
            # Get key by fingerprint
            ssh_key = await self._get_key_by_fingerprint(fingerprint)
            if not ssh_key:
                return False
            
            # Check key state
            if ssh_key.state != KeyState.ACTIVE:
                return False
            
            # Check expiration
            if ssh_key.expires_at and time.time() > ssh_key.expires_at:
                # Mark key as expired
                ssh_key.state = KeyState.EXPIRED
                await self._store_ssh_key_metadata(ssh_key)
                return False
            
            # Check IP restrictions
            if ssh_key.allowed_ips and source_ip not in ssh_key.allowed_ips:
                logger.warning("SSH key usage from unauthorized IP", 
                              key_id=ssh_key.id, source_ip=source_ip,
                              allowed_ips=ssh_key.allowed_ips)
                return False
            
            # Check rate limiting
            if not await self._check_usage_rate_limit(ssh_key.id, source_ip):
                return False
            
            # Update usage statistics
            ssh_key.usage_count += 1
            ssh_key.last_used_at = time.time()
            await self._store_ssh_key_metadata(ssh_key)
            
            # Log usage event
            await self._log_key_event(
                ssh_key.id, ssh_key.user_id, "used", source_ip, 
                "git", repository, True
            )
            
            # Cache result
            self.validation_cache[cache_key] = {
                'valid': True,
                'timestamp': time.time()
            }
            
            return True
            
        except Exception as e:
            logger.error("SSH key validation failed", 
                        fingerprint=fingerprint, error=str(e))
            return False

    async def get_user_private_key_path(self, user_id: str) -> Optional[str]:
        """Get path to user's primary private key for Git operations"""
        try:
            user_keys = await self.list_user_keys(user_id)
            active_keys = [
                key for key in user_keys 
                if key.state == KeyState.ACTIVE and key.is_generated
                and (not key.expires_at or key.expires_at > time.time())
            ]
            
            if not active_keys:
                return None
            
            # Sort by last used and select most recent
            active_keys.sort(key=lambda k: k.last_used_at or 0, reverse=True)
            primary_key = active_keys[0]
            
            # Get private key path
            private_key_path = self.key_storage_path / f"{primary_key.id}_private"
            
            if private_key_path.exists():
                return str(private_key_path)
            
            return None
            
        except Exception as e:
            logger.error("Failed to get private key path", 
                        user_id=user_id, error=str(e))
            return None

    # Helper methods for key operations
    def _calculate_fingerprint(self, public_key_bytes: bytes) -> str:
        """Calculate SSH key fingerprint using SHA256"""
        digest = hashes.Hash(hashes.SHA256())
        digest.update(public_key_bytes)
        hash_bytes = digest.finalize()
        
        # Format as SHA256 fingerprint
        b64_hash = base64.b64encode(hash_bytes).decode('ascii').rstrip('=')
        return f"SHA256:{b64_hash}"

    def _generate_key_id(self, user_id: str, fingerprint: str) -> str:
        """Generate unique key ID"""
        base_string = f"{user_id}:{fingerprint}:{time.time()}"
        hash_digest = hashlib.sha256(base_string.encode()).hexdigest()[:12]
        return f"key_{hash_digest}"

    def _format_public_key(self, public_key_bytes: bytes, comment: str, user_id: str) -> str:
        """Format public key with comment"""
        key_str = public_key_bytes.decode('utf-8')
        if comment:
            return f"{key_str} {comment}"
        else:
            return f"{key_str} {user_id}@build-platform"

    async def _store_private_key(self, key_id: str, private_key) -> None:

                encoding=serialization.Encoding.PEM,

                encryption_algorithm=serialization.NoEncryption()
            )

            # Store encrypted key

            logger.error("Failed to store private key", key_id=key_id, error=str(e))
            raise

    async def _delete_private_key(self, key_id: str) -> None:
        """Securely delete private key file"""
        try:
            private_key_path = self.key_storage_path / f"{key_id}_private"
            if private_key_path.exists():
                # Overwrite file with random data before deletion
                file_size = private_key_path.stat().st_size
                random_data = secrets.token_bytes(file_size)

            logger.error("Failed to delete private key", key_id=key_id, error=str(e))
